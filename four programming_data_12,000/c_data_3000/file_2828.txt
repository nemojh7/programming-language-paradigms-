{
    int total = 0;

    while(total < len)

        n = send(s, buf+total, bytesleft, 0);
        if (n == -1)
            break;
        total += n;
        bytesleft -= n;
    }

    return n==-1?-1:0;

LineBreaker::LineBreaker(const string &s)
{
        // Copy the string as a character array.
        m_data = new char[s.length()+1];
        strcpy(m_data, s.c_str());

        // Find all the words.
        char last = ' ';
        list<char *> parts;
        for(char *scan = m_data; *scan; ++scan) {
                char curr = *scan;

                if(isspace(last) && !isspace(curr))
                        parts.push_back(scan);
                else if(!isspace(last) && isspace(curr))
                        *scan = '\0';

                last = curr;
        }

        // Allocate the array of pointers for exec, and copy the
        // pointer into it.  Then add the NULL terminator.
        m_parmlist = new char * [parts.size()+1];
        copy(parts.begin(), parts.end(), m_parmlist);
        m_parmlist[parts.size()] = NULL;
}

$ /usr/bin/time sleep 1 | sleep 2
0.00user 0.00system 0:01.00elapsed 0%CPU (0avgtext+0avgdata 7488maxresident)k

As a side note, this program will produce syntax errors.What you have in every non-empty file is a series of bytes. Therefor, what your C program has to do is read bytes. Since the variable type char is used to represent a byte, and since you want to read multiple bytes at once for efficiency purposes, you read an array of chars. That's for the general understanding of what reading from a file means.
Going back to your example, there is no string type in C. A string is an array of bytes (char[]) ended by a nul character.
What the lecturer is doing is:

define an array of chars (which will contain a string) => char name[10] is an array that may contain a string of at most 9 characters (the last byte would be used for '\0').
ask fscanf to read a string, which means it will read multiple bytes (multiple chars) until it finds a nul character ('\0') and put all of that in the given array.

To understand what's happening, forget about string as an opaque data type (which might be true in other programming languages) and see them as what they really are: arrays of chars.
