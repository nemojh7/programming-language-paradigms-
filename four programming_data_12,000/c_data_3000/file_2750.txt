If you use Windows, x86-64 uses rcx, rdx, r8, and r9 for the four first (integer or pointer) arguments, in that order, and stack for the rest. {

}

{
    func(0, 1LL, 2LL);
}

        movl    $1, %esi
        xorl    %edi, %edi
