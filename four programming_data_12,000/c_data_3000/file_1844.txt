public static String asPoly(BigInteger b) {
    // Using BigInteger because it has a bitLength.
    List<String> parts = new ArrayList<>();
    for (int p = b.bitLength(); p >= 0; p--) {
        if (b.testBit(p)) {
            switch (p) {
                default:
                    parts.add("x^" + p);
                    break;
                case 1:
                    parts.add("x");
                    break;
                case 0:
                    parts.add("1");
                    break;
            }
        }
    }
    return String.join(" + ",parts);
