{

static void
wait_for(int fd, const char *expect, size_t siz)
{
        int rc = 0;
        char buf[1024];
        char *a = buf;
        assert(siz < sizeof buf);
        char *end = buf + siz;

        while( a < end ) {
                ssize_t s = read(fd, a, end - a);
                char *e = a + s;
                rc += 1;
                while( a < e && *a++ == *expect++ ) {
                        ;
                }
                if( s < 1 || a < e ) {
                        fprintf(stderr, "Ivalid data\nReceived: ");
                        show_data(buf, e - buf);
                        fprintf(stderr, "\nExpected: ");
                        show_data(expect, siz);
                        fputc('\n', stderr);
                        exit(1);
                }
        }
}
