def quantization_aware_training(model_path, save, w_bits, a_bits, symmetric, per_axis, narrow_range, quantizer, batch_size=64, epochs=2):
    """ Performs quantization aware training for 1 epoch on a model

    Args:
        model_path (string): Path to model to be quantized
        save (bool): Whether to save the model or not
        evaluate (bool): Whether to also evaluate the quantized model or not
        w_bits (int): Bit-width of quantized weights
        a_bits (int): Bit-width of quantized activations
        symmetric (bool): if true: symmetric quantization. Else: asymmetric quantization
        per_axis (bool): if true: per-channel quantization. Else: per-tensor quantization
        narrow_range (bool): if true: range is narrow, e.g. [-127, 127] for 8 bit. Else: range is full, e.g. [-128, 127] for 8 bit
        quantizer (tfmot.quantization.keras.quantizers): Quantizer method for determining floating point ranges (Only for weights. Activations always use MA).

            (TODO CURRENTLY NOT WORKING)

        q_aware_model (keras.models): Quantization aware model. Note that the model is not quantized, but only quantization aware.
        quant_accuracy (float): Accuracy of the quantized model when evaluated on the CIFAR-10 test set.
    """

    # Load model
    model = keras.models.load_model(model_path)

    # Create quantized model's name string
    name = model.name + '_'
    name = name + str(w_bits) + 'wbits_' + str(a_bits) + 'abits_'

    if symmetric:
        name = name + 'sym_'
    else:
        name = name + 'asym_'

    if narrow_range:
        name = name + 'narr_'
    else:
        name = name + 'full_'

    if per_axis:
        name = name + 'perch_'
    else:
        name = name + 'perten_'

    if quantizer == QAT_ALL_VALUES:
        name = name + 'AV'
    elif quantizer == QAT_LAST_VALUE:
        name = name + 'LV'
    elif quantizer == QAT_MA:
        name = name + 'MA'

    # *****
    quantize_apply = tfmot.quantization.keras.quantize_apply
    quantize_model = tfmot.quantization.keras.quantize_model
    quantize_annotate_layer = tfmot.quantization.keras.quantize_annotate_layer
    clone_model = tf.keras.models.clone_model
    quantize_scope = tfmot.quantization.keras.quantize_scope

    supported_layers = [
        tf.keras.layers.Conv2D,
    ]

    class Quantizer(tfmot.quantization.keras.QuantizeConfig):
        # Configure how to quantize weights.
        def get_weights_and_quantizers(self, layer):
            return [(layer.kernel, tfmot.quantization.keras.quantizers.LastValueQuantizer(num_bits=8, symmetric=True, narrow_range=False, per_axis=False))]

        # Configure how to quantize activations.
        def get_activations_and_quantizers(self, layer):
            return [(layer.activation, tfmot.quantization.keras.quantizers.MovingAverageQuantizer(num_bits=8, symmetric=False, narrow_range=False, per_axis=False))]

        def set_quantize_weights(self, layer, quantize_weights):

            # , in the same order
            layer.kernel = quantize_weights[0]

        def set_quantize_activations(self, layer, quantize_activations):
            # Add this line for each item returned in `get_activations_and_quantizers`
            # , in the same order.
            layer.activation = quantize_activations[0]

        # Configure how to quantize outputs (may be equivalent to activations).
        def get_output_quantizers(self, layer):
            return []

        def get_config(self):
            return {}

    class ConvQuantizer(Quantizer):
        # Configure weights to quantize with 4-bit instead of 8-bits.
        def get_weights_and_quantizers(self, layer):
            return [(layer.kernel, quantizer(num_bits=w_bits, symmetric=symmetric, narrow_range=narrow_range, per_axis=per_axis))]

        # Configure how to quantize activations.
        def get_activations_and_quantizers(self, layer):
            return [(layer.activation, tfmot.quantization.keras.quantizers.MovingAverageQuantizer(num_bits=a_bits, symmetric=False, narrow_range=False, per_axis=False))]

    class DepthwiseQuantizer(Quantizer):
        # Configure weights to quantize with 4-bit instead of 8-bits.
        def get_weights_and_quantizers(self, layer):
